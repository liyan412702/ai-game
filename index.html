const response = await fetch('https://api.deepseek.com/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${process.env.DEEPSEEK_API_KEY}` }, body: JSON.stringify({ model: 'deepseek-chat', messages: [ { role: 'system', content: `你是一位资深游戏架构师（如同小岛秀夫或宫本茂级别）。在生成配置前，你必须进行深度的“游戏拆解分析”。 【强制思考流程】（必须体现在 design_thoughts 字段中）： 1. 记忆检索：用户提到的游戏（如魂斗罗、塔防、跑酷等），它的核心乐趣是什么？玩家在玩的时候处于什么心理状态？ 2. 难度曲线：为了制造这种乐趣，数值该怎么调？（比如：魂斗罗必须充满紧张感，怪物要多；休闲跑酷必须放松）。 3. 参数映射：将你推导出的数值，精准填入 JSON 中。 【硬核设计指南】： - 射击类（如魂斗罗）：必须有高密度的敌人（enemyCount > 20），高速移动（enemySpeed > 180），给予玩家极强的压迫感。 - 塔防类：必须考验策略，怪物血量递增（enemyHp > 150），波数拉长（waves > 8）。 - 跑酷类：强调速度感，移速必须快（speed > 20）。 输出严格的 JSON 格式： { "design_thoughts": "在这里写你的拆解分析（一句话）", "gameType": "parkour|tower|dressup|shooter", "theme": "default|cyber|forest|desert|snow|space", "title": "酷炫且有内味的游戏名", "speed": 18, "density": 1.0, "targetScore": 300, "waves": 10, "enemyHp": 150, "coins": 300, "enemyCount": 25, "enemySpeed": 180, "playerHp": 100, "hairColor": "#ff0066", "clothesColor": "#00ffff", "pantsColor": "#ffcc00" } 注意：只输出 JSON，不要输出任何解释文字！` }, { role: 'user', content: prompt } ], response_format: { type: 'json_object' }, max_tokens: 500, temperature: 1.0 // 调到 1.0，释放它的狂野想象力 }) }); if (!response.ok) { const err = await response.text(); return new Response(JSON.stringify({ error: 'DeepSeek调用失败', detail: err }), { status: 502, headers: { 'Content-Type': 'application/json' } }); } const data = await response.json(); const content = data.choices?.[0]?.message?.content; if (!content || content.trim() === '') return new Response(JSON.stringify({ error: 'AI返回空内容' }), { status: 502, headers: { 'Content-Type': 'application/json' } }); const config = JSON.parse(content); // 将 AI 的“拆解思路”透传给前端，让你看看它的“学习成果” return new Response(JSON.stringify(config), { status: 200, headers: { 'Content-Type': 'application/json' } }); } catch (e) { return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: { 'Content-Type': 'application/json' } }); } }
